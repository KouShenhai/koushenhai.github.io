    FRP内网穿透

    你好呀，我的老朋友！我是老寇，欢迎来到老寇IoT云平台！

    # AMD64安装FRP

    # 切换root
sudo su

# 自行创建目录
mkdir -p /home/ubuntu/app

# AMD64
sudo wget https://github.com/fatedier/frp/releases/download/v0.63.0/frp_0.63.0_linux_amd64.tar.gz

# 解压
tar -zxvf frp_0.63.0_linux_amd64.tar.gz

# 移动
mv frp_0.63.0_linux_amd64 /usr/local/frp

# 进入frp目录
cd /usr/local/frp
    # FRP-SERVER配置
    vim /usr/local/frp/frps.toml

    # frps 监听地址
bindAddr = "0.0.0.0"
# frpc 与 frps 建立控制连接时使用的端口
bindPort = 7000
# frps Dashboard 管理后台的监听地址
webServer.addr = "0.0.0.0"
# frps Dashboard 管理后台端口
webServer.port = 7500
# Dashboard 登录用户名
webServer.user = "admin"
# Dashboard 登录密码
webServer.password = "123456"
# HTTPS 类型代理统一监听端口
vhostHTTPSPort = 443
# HTTP 类型代理统一监听端口
vhostHTTPPort = 80
    # FRP-CLIENT配置
    vim /usr/local/frp/frpc.toml

    serverAddr = "192.168.1.100"
serverPort = 7000

[[proxies]]
name = "1443_https"
type = "tcp"
localIp = "127.0.0.1"
localPort = 443
remotePort = 1443

[[proxies]]
name = "1122_ssh"
type = "tcp"
localIp = "127.0.0.1"
localPort = 22
remotePort = 1122
    # FRP-SERVER开机自启动
    vim /etc/systemd/system/frps.service

    [Unit]
Description=frps

[Service]
User=root
Group=root
Type=simple
Restart=always
RestartSec=3
AmbientCapabilities=CAP_NET_RAW
ExecStart=/usr/local/frp/frps -c /usr/local/frp/frps.toml
TimeoutStopSec=10
KillMode=mixed

[Install]
WantedBy=multi-user.target

    chmod -R 7777 /etc/systemd/system/frps.service
systemctl daemon-reload
systemctl enable frps
systemctl start frps
systemctl status frps
    # FRP-CLIENT开机自启动
    vim /etc/systemd/system/frpc.service

    [Unit]
Description=frpc

[Service]
User=root
Group=root
Type=simple
Restart=always
RestartSec=3
AmbientCapabilities=CAP_NET_RAW
ExecStart=/usr/local/frp/frpc -c /usr/local/frp/frpc.toml
TimeoutStopSec=10
KillMode=mixed

[Install]
WantedBy=multi-user.target

    chmod -R 7777 /etc/systemd/system/frpc.service
systemctl daemon-reload
systemctl enable frpc
systemctl start frpc
systemctl status frpc
